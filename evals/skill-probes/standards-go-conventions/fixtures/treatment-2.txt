package port

import (
	"fmt"
	"strconv"
)

func ParsePort(s string) (int, error) {
	port, err := strconv.Atoi(s)
	if err != nil {
		return 0, fmt.Errorf("parsing port %q: %w", s, err)
	}
	if port < 1 || port > 65535 {
		return 0, fmt.Errorf("validating port %d: must be between 1 and 65535", port)
	}
	return port, nil
}
