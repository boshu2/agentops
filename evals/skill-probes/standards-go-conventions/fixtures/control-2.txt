package port

import (
	"fmt"
	"strconv"
)

func ParsePort(s string) (int, error) {
	if s == "" {
		return 0, fmt.Errorf("invalid port %q", s)
	}

	for _, r := range s {
		if r < '0' || r > '9' {
			return 0, fmt.Errorf("invalid port %q", s)
		}
	}

	n, err := strconv.ParseUint(s, 10, 16)
	if err != nil || n < 1 {
		return 0, fmt.Errorf("invalid port %q", s)
	}

	return int(n), nil
}
