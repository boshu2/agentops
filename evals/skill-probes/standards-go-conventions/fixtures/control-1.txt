package port

import (
	"fmt"
	"strconv"
)

func ParsePort(s string) (int, error) {
	n, err := strconv.Atoi(s)
	if err != nil || n < 1 || n > 65535 {
		return 0, fmt.Errorf("invalid TCP port %q", s)
	}
	return n, nil
}
