package port

import (
	"fmt"
	"strconv"
)

func ParsePort(s string) (int, error) {
	port, err := strconv.Atoi(s)
	if err != nil {
		return 0, fmt.Errorf("invalid port %q: %w", s, err)
	}
	if port < 1 || port > 65535 {
		return 0, fmt.Errorf("port %q out of range", s)
	}
	return port, nil
}
